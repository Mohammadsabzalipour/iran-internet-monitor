function renderLineChart(canvasId, dataArray, lineColor) { const ctx = document.getElementById(canvasId).getContext('2d'); // محاسبه زمان شروع (۲۴ ساعت پیش) و ساخت لیبل‌های زمانی دقیق const now = new Date(); const startTime = new Date(now.getTime() - (24 * 60 * 60 * 1000)); const totalPoints = dataArray.length; const labels = dataArray.map((_, i) => { // محاسبه زمان برای هر نقطه بر اساس فاصله زمانی const pointTime = new Date(startTime.getTime() + (i * (24 * 60 * 60 * 1000 / totalPoints))); // فرمت ساعت به صورت "ساعت:دقیقه" const hh = pointTime.getHours().toString().padStart(2, '0'); const mm = pointTime.getMinutes().toString().padStart(2, '0'); return `${hh}:${mm}`; }); new Chart(ctx, { type: 'line', data: { labels: labels, datasets: [{ label: canvasId === 'bgpChart' ? 'BGP' : 'Ping', data: dataArray, borderColor: lineColor, backgroundColor: lineColor + '22', borderWidth: 2, fill: true, pointRadius: 0, // نقطه‌ها را مخفی نگه می‌داریم pointHoverRadius: 5, // وقتی موس روی خط رفت، نقطه نمایش داده شود pointHoverBackgroundColor: lineColor, tension: 0.4 }] }, options: { responsive: true, maintainAspectRatio: false, interaction: { mode: 'index', intersect: false, // باعث می‌شود حتی با نزدیک شدن به خط، تولتیپ باز شود }, plugins: { legend: { display: false }, tooltip: { backgroundColor: '#1e293b', titleColor: '#38bdf8', bodyColor: '#e2e8f0', borderColor: '#334155', borderWidth: 1, padding: 10, displayColors: false, // حذف مربع رنگی کنار متن callbacks: { // شخصی‌سازی متن تولتیپ label: function(context) { return 'مقدار: ' + context.parsed.y; }, title: function(context) { return 'ساعت: ' + context[0].label; } } } }, scales: { x: { display: true, // محور افقی را نمایش بده grid: { color: '#334155', drawBorder: false }, ticks: { color: '#94a3b8', font: {size: 10, family: 'Vazirmatn'}, maxTicksLimit: 6 // فقط ۶ ساعت مختلف را نشان بده تا شلوغ نشود } }, y: { display: true, grid: { color: '#334155' }, ticks: { color: '#94a3b8', font: {size: 10} } } } } }); }
